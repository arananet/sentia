Identity is governed by brand/BRAND.md. This system applies that identity to

## Tokens

Paper #FAFAF6 occupies roughly 90 percent of the surface. Ink green #12211C
forms text and instrument outlines. Signal teal #0E7C66 marks links, labels
and the Find My network and LoRaWAN paths. Coral #FF6B4A is reserved for the
status LED and alert dots. Soft mint #EFF4EE is a rare panel fill; hairline

## Type scale

Space Grotesk at weight 300 is the display face. Display type must serve the
product. The hero runs from 56 to 64 px with 0.96 leading and -0.055 em tracking. Section titles run from
40 to 64 px with 1.04 leading and -0.04 em tracking, limited to 18 characters
per line where space permits. Data values run from 32 to 56 px. Inter body
text is 17 px with 1.65 leading; introductory text is 18–20 px, at most
54 characters per line. System mono labels are 11 px, uppercase, with
0.16 em tracking. Never shrink body copy to fit a composition.

## Spacing rhythm

Use an 8 px base: 8, 16, 24, 32, 48, 64, 96, 128, 160. The page container
is at most 1440 px wide, with fluid 24–80 px gutters. Sections have 64–96 px
vertical padding for richer density. A title, its introduction and its
illustration form one idea; detailed lists follow with 32–48 px separation. Hairlines provide
structure without enclosing every item in a card. Technical examples and
tables remain dense enough to scan, with scrolling contained locally.
Fixed margins, padding and gaps use multiples of 8 px, including the bounds
of fluid spacing. One-pixel border overlaps and diagram coordinates are
optical geometry rather than content spacing.

function parseGlb(buf) {
  assert.equal(buf.readUInt32LE(0), 0x46546c67, 'GLB magic "glTF"');
  assert.equal(buf.readUInt32LE(8), buf.length, 'GLB header length equals byte length');
  const jlen = buf.readUInt32LE(12);
  assert.equal(buf.readUInt32LE(16), 0x4e4f534a, 'first chunk is JSON');
  return JSON.parse(buf.toString('utf8', 20, 20 + jlen));
}
async function moduleInfo() {

// World bounds of a subtree from POSITION accessor min and max (nodes carry translation only here;
// the frozen product test in take-apart-product-model keeps the full matrix reader).
function worldBounds(gltf, idx, off = [0, 0, 0]) {
  const n = gltf.nodes[idx];
  const t = n.translation || [0, 0, 0];
  const o = [off[0] + t[0], off[1] + t[1], off[2] + t[2]];
  const lo = [Infinity, Infinity, Infinity], hi = [-Infinity, -Infinity, -Infinity];
  const add = (l, h) => { for (let i = 0; i < 3; i++) { lo[i] = Math.min(lo[i], l[i]); hi[i] = Math.max(hi[i], h[i]); } };
  if (n.mesh != null) {
    assert.ok(!n.rotation && !n.scale && !n.matrix, 'descendant nodes are plain translations');
    for (const p of gltf.meshes[n.mesh].primitives) {
      const a = gltf.accessors[p.attributes.POSITION];
      add(a.min.map((v, i) => v + o[i]), a.max.map((v, i) => v + o[i]));
    }
  }
  for (const c of n.children || []) { const b = worldBounds(gltf, c, o); if (b) add(b.lo, b.hi); }
  return lo[0] === Infinity ? null : { lo, hi };
}

test('web-render-finish S2: five FACTS nodes in stack order at layerCenterMm(t=0) within 0.01, 36 mm footprint, 2.5 mm profile', async () => {
  const { bytes } = await moduleInfo();
  const gltf = parseGlb(bytes);
  const top = gltf.scenes[gltf.scene ?? 0].nodes;
  assert.deepEqual(top.map((i) => gltf.nodes[i].name), NAMES, 'exactly the five FACTS names in stack order');
  top.forEach((idx, k) => {
    const t = gltf.nodes[idx].translation || [0, 0, 0];
    assert.ok(Math.abs(t[0]) <= 0.01 && Math.abs(t[2]) <= 0.01, `${NAMES[k]} x and z are 0`);
    assert.ok(Math.abs(t[1] - M.layerCenterMm(k, 0)) <= 0.01, `${NAMES[k]} y ${t[1]} vs layerCenterMm ${M.layerCenterMm(k, 0)}`);
  });
  const lo = [Infinity, Infinity, Infinity], hi = [-Infinity, -Infinity, -Infinity];
  for (const idx of top) {
    const b = worldBounds(gltf, idx);
    for (let a = 0; a < 3; a++) { lo[a] = Math.min(lo[a], b.lo[a]); hi[a] = Math.max(hi[a], b.hi[a]); }
  }
  assert.ok(Math.abs(hi[0] - lo[0] - 36) <= 0.5, `footprint x ${(hi[0] - lo[0]).toFixed(3)}`);
  assert.ok(Math.abs(hi[2] - lo[2] - 36) <= 0.5, `footprint z ${(hi[2] - lo[2]).toFixed(3)}`);
  assert.ok(Math.abs(hi[1] - lo[1] - 2.5) <= 0.1, `profile ${(hi[1] - lo[1]).toFixed(3)}`);
  // Layer thicknesses are frozen (local bounds of each layer, descendants only).
  top.forEach((idx, k) => {
    const n = gltf.nodes[idx];
    const kids = (n.children || []).map((c) => worldBounds(gltf, c));
    const l = Math.min(...kids.map((b) => b.lo[1])), h = Math.max(...kids.map((b) => b.hi[1]));
    const slack = k === 4 ? 0.3 : 0.1;
    assert.ok(h - l >= M.LAYERS[k].thicknessMm * 0.4 && h - l <= M.LAYERS[k].thicknessMm + slack, `${NAMES[k]} thickness ${(h - l).toFixed(3)}`);
  });
});
